+class OlympiadVariant(db.Model):
+    """Сгенерированный вариант олимпиады (набор из 5 задач)."""
+    __tablename__ = 'olympiad_variants'
+
+    id = db.Column(db.String(36), primary_key=True)  # uuid4
+    olympiad_slug = db.Column(db.String(100), nullable=False, index=True)
+    olympiad_title = db.Column(db.String(200), nullable=True)
+    round_key = db.Column(db.String(50), nullable=True)
+    round_title = db.Column(db.String(200), nullable=True)
+    grade = db.Column(db.Integer, nullable=False, index=True)
+    user_id = db.Column(db.Integer, db.ForeignKey('users.id'), nullable=True, index=True)
+    created_at = db.Column(db.DateTime, default=datetime.utcnow)
+
+    # Связи
+    user = db.relationship('User', backref=db.backref('olympiad_variants', lazy='dynamic'))
+    tasks = db.relationship('OlympiadTask', backref='variant',

+class OlympiadTask(db.Model):
+    """Одна задача в сгенерированном варианте олимпиады."""
+    __tablename__ = 'olympiad_tasks'
+
+    id = db.Column(db.Integer, primary_key=True)
+    variant_id = db.Column(db.String(36), db.ForeignKey('olympiad_variants.id',
+                           ondelete='CASCADE'), nullable=False, index=True)
+    position = db.Column(db.Integer, nullable=False)  # 1..5
+
+    # Тексты
+    text = db.Column(db.Text, nullable=False)              # финальный LaTeX
+    original_text = db.Column(db.Text, nullable=True)      # прототип (для отладки)
+    solution = db.Column(db.Text, nullable=True)            # полное решение
+    answer = db.Column(db.String(500), nullable=True)       # краткий ответ
+
+    # Статус пайплайна
+    status = db.Column(db.String(20), nullable=False, default='draft', index=True)
+    # Возможные значения: 'draft' | 'validated' | 'failed' | 'regenerating'
+    validation_errors = db.Column(db.Text, nullable=True)  # JSON список ошибок
+
+    # Метаданные
+    topic = db.Column(db.String(100), nullable=True, index=True)
+    source_year = db.Column(db.Integer, nullable=True)
+    source_problem = db.Column(db.Integer, nullable=True)
+    author = db.Column(db.String(200), nullable=True)
+    pipeline_version = db.Column(db.String(10), nullable=True, default='1.0')
+    validated_at = db.Column(db.DateTime, nullable=True)
+
+    # Связи
+    attempts = db.relationship('OlympiadTaskAttempt', backref='task',
+                               cascade='all, delete-orphan', order_by='OlympiadTaskAttempt.created_at')
+
+    def to_dict(self):
+        """Конвертация в словарь для JSON / шаблонов."""
+        return {
+            'id': self.id,
+            'position': self.position,
+            'text': self.text,
+            'answer': self.answer,
+            'solution': self.solution,
+            'topic': self.topic,
+            'status': self.status,
+            'original_text': self.original_text,
+        }
+
+    def __repr__(self):
+        return f'<OlympiadTask {self.id} pos={self.position} status={self.status} topic={self.topic}>'
+
